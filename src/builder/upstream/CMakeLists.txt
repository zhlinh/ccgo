cmake_minimum_required(VERSION 3.18)
project(${CCGO_UPSTREAM_NAME} LANGUAGES C CXX)

if(NOT EXISTS "${CCGO_UPSTREAM_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "build.cmake.source_dir must contain CMakeLists.txt: ${CCGO_UPSTREAM_SOURCE}")
endif()
if(NOT CCGO_UPSTREAM_TARGET)
    message(FATAL_ERROR "build.cmake.target is required when source_dir is set")
endif()
if(IS_ABSOLUTE "${CCGO_UPSTREAM_INCLUDE}" OR CCGO_UPSTREAM_INCLUDE MATCHES "(^|[/\\])\\.\\.([/\\]|$)")
    message(FATAL_ERROR "install_include_dir must stay inside the install prefix")
endif()
set(BUILD_SHARED_LIBS "${CCGO_BUILD_SHARED_LIBS}" CACHE BOOL "Build shared libraries" FORCE)
if(DEFINED CCGO_USER_CMAKE_FILES)
    foreach(script IN LISTS CCGO_USER_CMAKE_FILES)
        include("${script}")
    endforeach()
elseif(EXISTS "${CCGO_PROJECT_ROOT}/CCGO.cmake")
    include("${CCGO_PROJECT_ROOT}/CCGO.cmake")
endif()
add_subdirectory("${CCGO_UPSTREAM_SOURCE}" upstream)
if(NOT TARGET "${CCGO_UPSTREAM_TARGET}")
    message(FATAL_ERROR "Upstream library target does not exist: ${CCGO_UPSTREAM_TARGET}")
endif()
get_target_property(alias "${CCGO_UPSTREAM_TARGET}" ALIASED_TARGET)
if(alias)
    set(CCGO_UPSTREAM_TARGET "${alias}")
endif()
get_target_property(imported "${CCGO_UPSTREAM_TARGET}" IMPORTED)
get_target_property(kind "${CCGO_UPSTREAM_TARGET}" TYPE)
if(imported)
    message(FATAL_ERROR "build.cmake.target must be built from source")
endif()
set(sdk_output "${CMAKE_BINARY_DIR}/out")
if(CCGO_BUILD_SHARED_LIBS)
    if(NOT kind STREQUAL "SHARED_LIBRARY")
        message(FATAL_ERROR "Shared SDK requires a SHARED_LIBRARY target")
    endif()
    set_target_properties("${CCGO_UPSTREAM_TARGET}" PROPERTIES
        OUTPUT_NAME "${CCGO_UPSTREAM_NAME}" DEBUG_POSTFIX "" VERSION "" SOVERSION ""
        LIBRARY_OUTPUT_DIRECTORY "${sdk_output}"
        RUNTIME_OUTPUT_DIRECTORY "${sdk_output}" ARCHIVE_OUTPUT_DIRECTORY "${sdk_output}")
    foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties("${CCGO_UPSTREAM_TARGET}" PROPERTIES
            ${config}_POSTFIX ""
            LIBRARY_OUTPUT_DIRECTORY_${config} "${sdk_output}"
            RUNTIME_OUTPUT_DIRECTORY_${config} "${sdk_output}"
            ARCHIVE_OUTPUT_DIRECTORY_${config} "${sdk_output}")
    endforeach()
    add_custom_target(ccgo_upstream DEPENDS "${CCGO_UPSTREAM_TARGET}")
else()
    if(NOT kind STREQUAL "STATIC_LIBRARY")
        message(FATAL_ERROR "Static SDK requires a STATIC_LIBRARY target")
    endif()
    # Traverse linked targets, rather than merging every archive in the build.
    function(collect_archives target)
        if(NOT TARGET "${target}")
            if(target MATCHES "\\$<")
                message(FATAL_ERROR "Unsupported dependency expression: ${target}. Use a project adapter for this link graph.")
            endif()
            if(IS_ABSOLUTE "${target}" AND target MATCHES "\\.(a|lib)$")
                message(FATAL_ERROR "Archive dependencies must be CMake targets: ${target}")
            endif()
            return()
        endif()
        get_target_property(alias "${target}" ALIASED_TARGET)
        if(alias)
            set(target "${alias}")
        endif()
        get_property(visited GLOBAL PROPERTY CCGO_VISITED)
        if(target IN_LIST visited)
            return()
        endif()
        set_property(GLOBAL APPEND PROPERTY CCGO_VISITED "${target}")
        get_target_property(kind "${target}" TYPE)
        if(kind STREQUAL "STATIC_LIBRARY")
            set_property(GLOBAL APPEND PROPERTY CCGO_ARCHIVES "${target}")
        endif()
        foreach(property LINK_LIBRARIES INTERFACE_LINK_LIBRARIES)
            get_target_property(dependencies "${target}" "${property}")
            foreach(dependency IN LISTS dependencies)
                string(REGEX REPLACE "^\\$<LINK_ONLY:(.*)>$" "\\1" dependency "${dependency}")
                string(REGEX REPLACE "^\\$<BUILD_INTERFACE:(.*)>$" "\\1" dependency "${dependency}")
                collect_archives("${dependency}")
            endforeach()
        endforeach()
    endfunction()
    collect_archives("${CCGO_UPSTREAM_TARGET}")
    get_property(archives GLOBAL PROPERTY CCGO_ARCHIVES)
    set(inputs "")
    foreach(target IN LISTS archives)
        string(APPEND inputs "\"$<TARGET_FILE:${target}>\"\n")
    endforeach()
    if(MSVC)
        set(merged "${sdk_output}/${CCGO_UPSTREAM_NAME}.lib")
    else()
        set(merged "${sdk_output}/lib${CCGO_UPSTREAM_NAME}.a")
    endif()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/merge-$<CONFIG>.cmake" CONTENT
        "set(INPUTS\n${inputs})\nset(OUTPUT [==[${merged}]==])\nset(AR [==[${CMAKE_AR}]==])\nset(RANLIB [==[${CMAKE_RANLIB}]==])\nset(APPLE ${APPLE})\nset(MSVC ${MSVC})\ninclude([==[${CMAKE_CURRENT_LIST_DIR}/merge.cmake]==])\n")
    add_custom_target(ccgo_upstream
        COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_BINARY_DIR}/merge-$<CONFIG>.cmake"
        DEPENDS ${archives} VERBATIM)
endif()
configure_file(headers.cmake.in "${CMAKE_BINARY_DIR}/ccgo-headers.cmake" @ONLY)
install(SCRIPT "${CMAKE_BINARY_DIR}/ccgo-headers.cmake")
